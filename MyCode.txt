private static findProjectRoot(start: string): string {
        // Walk up to 12 levels looking for project markers.
        // Hard markers (build files) return immediately — most authoritative.
        // Soft markers (resources/, src/main/java, testng.xml, qaf-config*) are
        // recorded; the highest soft-marker ancestor wins if no hard marker
        // is found. This handles "code dump" legacy projects without pom.xml
        // (e.g. <root>/LegacySeleniumCodeForConversion/resources/<env>/...).
        let cur = start;
        let softMarkerAncestor: string | null = null;
        const HARD_MARKERS = ['pom.xml', 'build.gradle', 'build.gradle.kts', 'package.json'];
        const SOFT_MARKER_DIRS = ['resources', 'resource'];
        const SOFT_MARKER_FILES = /^(testng|qaf-config|cucumber\.runtime|application)[-_a-z0-9]*\.(xml|properties)$/i;
        for (let i = 0; i < 12; i++) {
            // 1. Hard build-file markers — return immediately.
            for (const c of HARD_MARKERS) {
                if (fs.existsSync(path.join(cur, c))) return cur;
            }
            // 2. .csproj / .sln files — C# project markers.
            try {
                const ents = fs.readdirSync(cur);
                if (ents.some((e) => e.endsWith('.csproj') || e.endsWith('.sln'))) return cur;
                // 3. Soft marker: <cur>/resources/ or <cur>/resource/ directory.
                //    Common in QAF / TestNG projects that ship data files under
                //    resources/<env>/testdata/. Record the HIGHEST such ancestor
                //    seen — `softMarkerAncestor` only overwrites going up.
                for (const dirName of SOFT_MARKER_DIRS) {
                    if (ents.includes(dirName)) {
                        try {
                            if (fs.statSync(path.join(cur, dirName)).isDirectory()) {
                                softMarkerAncestor = cur;
                            }
                        } catch { /* ignore */ }
                    }
                }
                // 4. Soft marker: testng.xml / qaf-config*.xml / cucumber.runtime.xml
                //    living directly in this directory.
                if (ents.some((e) => SOFT_MARKER_FILES.test(e))) {
                    softMarkerAncestor = cur;
                }
            } catch { /* ignore */ }
            const parent = path.dirname(cur);
            if (parent === cur) break;
            cur = parent;
        }
        if (softMarkerAncestor) return softMarkerAncestor;
        return start;
    }

/**
 * Find a data file by basename anywhere reachable from `root` (no env-segment
 * filter). Used as a softer fallback when env-pathed search fails (e.g. a
 * project that drops all data files under a single `testdata/` folder without
 * per-env subfolders).
 */
function findFileByBasename(
    root: string,
    targetBaseName: string,
    maxDepth = 12,
): string | null {
    const targetLower = targetBaseName.toLowerCase();
    const queue: Array<{ dir: string; depth: number }> = [{ dir: root, depth: 0 }];
    while (queue.length > 0) {
        const { dir, depth } = queue.shift()!;
        if (depth > maxDepth) continue;
        let entries: import('fs').Dirent[];
        try { entries = fs.readdirSync(dir, { withFileTypes: true }); }
        catch { continue; }
        for (const e of entries) {
            const full = path.join(dir, e.name);
            if (e.isFile() && e.name.toLowerCase() === targetLower) {
                return full;
            }
            if (e.isDirectory() && !e.name.startsWith('.') &&
                e.name !== 'node_modules' && e.name !== 'dist' && e.name !== 'build') {
                queue.push({ dir: full, depth: depth + 1 });
            }
        }
    }
    return null;
}

/**
 * When the inferred workspaceRoot is too deep (project-root detection
 * missed the real ancestor that contains `resources/`), walk UP from
 * `start` and return the highest ancestor that has either a `resources/`
 * or `resource/` directory. If none seen within `maxLevels`, return `start`.
 */
function findResourcesAncestor(start: string, maxLevels = 8): string {
    let cur = start;
    let highest = start;
    for (let i = 0; i < maxLevels; i++) {
        try {
            const ents = fs.readdirSync(cur);
            if (ents.includes('resources') || ents.includes('resource')) {
                highest = cur;
            }
        } catch { /* ignore */ }
        const parent = path.dirname(cur);
        if (parent === cur) break;
        cur = parent;
    }
    return highest;
}
